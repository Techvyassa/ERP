# XAMPP Virtual Host Configuration
# Copy this content to: C:\xampp\apache\conf\extra\httpd-vhosts.conf
# Replace the path with your actual Laravel installation path

# Main domain and all subdomains for Laravel
<VirtualHost *:80>
    ServerName localhost
    ServerAlias *.localhost
    DocumentRoot "C:/xampp/htdocs/erp/ERP/public"
    
    <Directory "C:/xampp/htdocs/erp/ERP/public">
        Options Indexes FollowSymLinks MultiViews
        AllowOverride All
        Order allow,deny
        Allow from all
        Require all granted
    </Directory>
    
    ErrorLog "logs/laravel-error.log"
    CustomLog "logs/laravel-access.log" common
</VirtualHost>
